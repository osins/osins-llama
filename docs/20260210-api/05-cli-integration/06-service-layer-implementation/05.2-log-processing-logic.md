# 日志处理逻辑

## 概述

日志处理逻辑负责处理CLI的日志记录、过滤和输出功能。

## 日志处理器

```python
import logging
from pathlib import Path
import re
import json
from datetime import datetime
from typing import Callable, Optional
import threading


class LogProcessor:
    """日志处理器"""
    
    def __init__(self, logger: logging.Logger = None):
        self.logger = logger or logging.getLogger(__name__)
    
    def tail_log_file(self, log_file: Path, lines: int = 50) -> str:
        """获取日志文件最后几行"""
        if not log_file.exists():
            return f"Log file does not exist: {log_file}"
        
        try:
            with open(log_file, 'r', encoding='utf-8') as f:
                all_lines = f.readlines()
                last_lines = all_lines[-lines:] if len(all_lines) >= lines else all_lines
            
            return ''.join(last_lines)
        except Exception as e:
            self.logger.error(f"Error reading log file {log_file}: {str(e)}")
            return f"Error reading log file: {str(e)}"
    
    def filter_sensitive_info(self, log_line: str) -> str:
        """过滤敏感信息"""
        # 过滤API密钥
        api_key_pattern = r'(sk-[a-zA-Z0-9]{10,})'
        filtered_line = re.sub(api_key_pattern, '***REDACTED***', log_line)
        
        # 过滤其他敏感信息
        filtered_line = re.sub(r'(?i)(password|token|key)\s*[:=]\s*\S+', r'\1=***REDACTED***', filtered_line)
        
        return filtered_line
    
    def follow_log_file(self, log_file: Path, callback: Callable[[str], None], stop_event: threading.Event = None, buffer_size: int = 1024):
        """实时跟踪日志文件"""
        if not log_file.exists():
            self.logger.warning(f"Log file does not exist: {log_file}")
            return
        
        with open(log_file, 'r', encoding='utf-8') as f:
            # 移动到文件末尾
            f.seek(0, 2)
            
            while not (stop_event and stop_event.is_set()):
                line = f.readline()
                if line:
                    filtered_line = self.filter_sensitive_info(line.rstrip())
                    callback(filtered_line)
                else:
                    import time
                    time.sleep(0.1)
    
    def setup_rotating_log(self, log_file: Path, max_bytes: int = 10*1024*1024, backup_count: int = 5) -> logging.Logger:
        """设置轮转日志"""
        from logging.handlers import RotatingFileHandler
        
        log_file.parent.mkdir(parents=True, exist_ok=True)
        
        logger = logging.getLogger("cli_logger")
        logger.setLevel(logging.INFO)
        
        # 防止重复添加处理器
        if logger.handlers:
            logger.handlers.clear()
        
        handler = RotatingFileHandler(
            str(log_file),
            maxBytes=max_bytes,
            backupCount=backup_count
        )
        
        formatter = logging.Formatter('%(asctime)s - %(name)s - %(levelname)s - %(message)s')
        handler.setFormatter(formatter)
        logger.addHandler(handler)
        
        return logger
    
    def log_structured_data(self, logger: logging.Logger, level: str, message: str, **kwargs):
        """记录结构化数据"""
        log_entry = {
            'timestamp': datetime.now().isoformat(),
            'level': level.upper(),
            'message': message,
            **kwargs
        }
        
        # 示例输出: {"timestamp": "2023-10-20T10:30:00.123456", "level": "INFO", "message": "Server started", "port": 31301}
        log_method = getattr(logger, level.lower(), logger.info)
        log_method(json.dumps(log_entry))
    
    def sanitize_log_data(self, data) -> dict:
        """净化日志数据，移除敏感信息"""
        if isinstance(data, dict):
            sanitized = {}
            for key, value in data.items():
                if isinstance(value, (dict, list)):
                    sanitized[key] = self.sanitize_log_data(value)
                elif isinstance(key, str) and any(sensitive in key.lower() for sensitive in ['password', 'token', 'key', 'secret']):
                    sanitized[key] = "***REDACTED***"
                elif isinstance(value, str) and any(sensitive in value.lower() for sensitive in ['password', 'token', 'key', 'secret']):
                    sanitized[key] = "***REDACTED***"
                else:
                    sanitized[key] = value
            return sanitized
        elif isinstance(data, list):
            return [self.sanitize_log_data(item) for item in data]
        else:
            return data
```
